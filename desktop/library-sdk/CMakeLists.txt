cmake_minimum_required(VERSION 3.20)
project(spklib_windows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
file(STRINGS VERSION SPKLIB_VERSION LIMIT_COUNT 1)
add_library(spklib STATIC src/api.cpp src/library.cpp src/capture_one.cpp src/sqlite.cpp)
target_include_directories(spklib PUBLIC include src third_party)
target_compile_definitions(spklib PRIVATE SPKLIB_VERSION="${SPKLIB_VERSION}")
if(WIN32)
  target_compile_options(spklib PUBLIC /utf-8 /FI${CMAKE_CURRENT_SOURCE_DIR}/src/windows_compat.hpp)
  target_compile_definitions(spklib PUBLIC NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(spklib PUBLIC winsqlite3 normaliz)
else()
  find_package(SQLite3 REQUIRED)
  target_link_libraries(spklib PUBLIC SQLite::SQLite3)
  if(APPLE)
    target_link_libraries(spklib PUBLIC "-framework CoreFoundation")
  endif()
endif()
add_executable(spklib-host host/main.cpp)
add_executable(spklib-tests tests/tests.cpp)
target_link_libraries(spklib-host PRIVATE spklib)
target_link_libraries(spklib-tests PRIVATE spklib)
target_compile_definitions(spklib-tests PRIVATE SPKLIB_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
if(WIN32)
  foreach(app spklib-host spklib-tests)
    target_link_options(${app} PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/windows.manifest)
  endforeach()
endif()
enable_testing()
add_test(NAME spklib-sdk COMMAND spklib-tests)
